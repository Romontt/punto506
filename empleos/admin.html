// Función para ver/ocultar la clave function toggleClave() { const input = document.getElementById('clave-maestra'); input.type = input.type === 'password' ? 'text' : 'password'; } // Función principal de Login conectada a Supabase async function intentarLogin() { const intento = document.getElementById('clave-maestra').value; const errorCont = document.getElementById('error-container'); const btn = document.querySelector('button[onclick="intentarLogin()"]'); if (!intento) return; // Feedback visual de carga btn.innerText = "VERIFICANDO..."; btn.disabled = true; try { // Llamamos a la función RPC de Supabase // clave_intento es el nombre del parámetro que definiste en el SQL const { data: esValido, error } = await supabase.rpc('login_admin', { clave_intento: intento }); if (error) throw error; if (esValido === true) { // ÉXITO: Guardamos el estado en la sesión sessionStorage.setItem('key_admin_506', intento); // Animación de salida document.getElementById('gatekeeper').style.opacity = '0'; setTimeout(() => { document.getElementById('gatekeeper').style.display = 'none'; const dash = document.getElementById('dashboard'); dash.classList.remove('hidden'); setTimeout(() => dash.classList.add('opacity-100'), 50); }, 500); } else { // FALLO: La base de datos dijo que no mostrarError(); } } catch (err) { console.error("Error de autenticación:", err); mostrarError(); } finally { btn.innerText = "VERIFICAR IDENTIDAD"; btn.disabled = false; } } function mostrarError() { const errorCont = document.getElementById('error-container'); errorCont.classList.remove('hidden'); document.getElementById('clave-maestra').value = ''; // Animación de sacudida en el cuadro const glass = document.querySelector('.glass'); glass.classList.add('animate-shake'); setTimeout(() => glass.classList.remove('animate-shake'), 500); } function logout() { sessionStorage.clear(); window.location.reload(); } // Al cargar, verificamos si ya había una sesión activa document.addEventListener('DOMContentLoaded', async () => { const savedKey = sessionStorage.getItem('key_admin_506'); if (savedKey) { // Validamos la llave guardada para estar seguros const { data } = await supabase.rpc('login_admin', { clave_intento: savedKey }); if (data === true) { document.getElementById('gatekeeper').style.display = 'none'; const dash = document.getElementById('dashboard'); dash.classList.remove('hidden'); dash.classList.add('opacity-100'); } else { sessionStorage.clear(); } } });
