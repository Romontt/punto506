// Lógica para cargar solicitudes en admin-empleos.js async function cargarSolicitudes() { // Traemos solo los que NO están aprobados (asumiendo que tienes esa columna) // O simplemente traemos todos para gestionar const { data: empleos, error } = await supabase .from('empleos') .select('*') .order('created_at', { ascending: false }); const container = document.getElementById('solicitudes-container'); container.innerHTML = ''; empleos.forEach(job => { container.innerHTML += `

${job.nombre_comercio}

${job.titulo_puesto}

`; }); } async function aprobar(id) { const clave = sessionStorage.getItem('key_admin'); // Usamos tu función RPC 'aprobar_empleo' const { error } = await supabase.rpc('aprobar_empleo', { empleo_id: id, clave_admin: clave }); if(!error) cargarSolicitudes(); }
